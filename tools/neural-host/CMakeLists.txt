cmake_minimum_required(VERSION 3.20)
project(openomsi_neural_host LANGUAGES CXX)
option(OPENOMSI_ENABLE_NGX "Compile the explicitly experimental NGX Feature18 adapter (MSVC SDK ABI)" OFF)
enable_testing()
add_executable(neural-protocol-test protocol_test.cpp)
target_compile_features(neural-protocol-test PRIVATE cxx_std_17)
add_test(NAME neural-protocol COMMAND neural-protocol-test)
add_executable(neural-pe-test pe_test.cpp)
target_compile_features(neural-pe-test PRIVATE cxx_std_17)
add_test(NAME neural-pe COMMAND neural-pe-test)
if(NOT WIN32)
  add_executable(openomsi-neural-host main.cpp)
  target_compile_features(openomsi-neural-host PRIVATE cxx_std_17)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The neural host supports Windows x64 only")
  endif()
  if(NOT EXISTS "${STREAMLINE_SDK}/include/sl.h")
    message(FATAL_ERROR "Set STREAMLINE_SDK to an authorized Streamline 2.14.16 SDK")
  endif()
  add_executable(openomsi-neural-host main.cpp streamline.cpp)
  target_compile_features(openomsi-neural-host PRIVATE cxx_std_17)
  target_compile_definitions(openomsi-neural-host PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(openomsi-neural-host PRIVATE "${STREAMLINE_SDK}/include")
  target_link_libraries(openomsi-neural-host PRIVATE d3d12 dxgi wintrust crypt32 bcrypt user32 version)
  if(OPENOMSI_ENABLE_NGX)
    if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
      message(FATAL_ERROR "NGX official library requires native MSVC x64. MinGW can build diagnostics/Streamline only.")
    endif()
    if(NOT NGX_SDK)
      set(NGX_SDK "${STREAMLINE_SDK}/external/ngx-sdk")
    endif()
    unset(NGX_INCLUDE CACHE)
    unset(NGX_LIBRARY CACHE)
    find_path(NGX_INCLUDE nvsdk_ngx.h PATHS "${NGX_SDK}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(NGX_LIBRARY NAMES nvsdk_ngx_d PATHS "${NGX_SDK}/lib/Windows_x86_64" "${NGX_SDK}/lib/Windows_x86_64/x64" NO_DEFAULT_PATH REQUIRED)
    target_sources(openomsi-neural-host PRIVATE ngx.cpp)
    target_include_directories(openomsi-neural-host PRIVATE "${NGX_INCLUDE}")
    target_compile_definitions(openomsi-neural-host PRIVATE OPENOMSI_WITH_NGX=1)
    # NVIDIA's _d library uses the dynamic release CRT. Use it for all configurations.
    set_property(TARGET openomsi-neural-host PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    target_link_libraries(openomsi-neural-host PRIVATE "${NGX_LIBRARY}" shlwapi shell32 ole32 advapi32)
  endif()
  if(MINGW)
    target_link_options(openomsi-neural-host PRIVATE -static -municode)
    target_link_options(neural-protocol-test PRIVATE -static)
    target_link_options(neural-pe-test PRIVATE -static)
  endif()
  add_test(NAME neural-host-pe-imports COMMAND neural-pe-test $<TARGET_FILE:openomsi-neural-host>)
endif()
add_test(NAME neural-host-self-test COMMAND openomsi-neural-host --self-test)
